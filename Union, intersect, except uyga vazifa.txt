DROP TABLE IF EXISTS students;
DROP TABLE IF EXISTS teachers;

CREATE TABLE students (
    id INTEGER PRIMARY KEY,
    first_name TEXT,
    last_name TEXT
);

CREATE TABLE teachers (
    id INTEGER PRIMARY KEY,
    first_name TEXT,
    last_name TEXT
);

SELECT first_name, last_name FROM students 
WHERE first_name LIKE 'A%' OR first_name LIKE 'E%' OR first_name LIKE 'I%' OR first_name LIKE 'O%' OR first_name LIKE 'U%'
UNION
SELECT first_name, last_name FROM teachers 
WHERE first_name LIKE 'A%' OR first_name LIKE 'E%' OR first_name LIKE 'I%' OR first_name LIKE 'O%' OR first_name LIKE 'U%';

SELECT * FROM (
    SELECT first_name, last_name FROM students 
    UNION 
    SELECT first_name, last_name FROM teachers
) 
ORDER BY length(last_name) ASC, last_name ASC;

SELECT * FROM (
    SELECT first_name, last_name, 'student' AS source FROM students 
    UNION
    SELECT first_name, last_name, 'teacher' AS source FROM teachers
) 
ORDER BY source, last_name ASC;

SELECT * FROM (
    SELECT first_name, last_name, id FROM students 
    UNION 
    SELECT first_name, last_name, id FROM teachers
) 
ORDER BY id DESC 
LIMIT 3;

DROP TABLE IF EXISTS orders;
DROP TABLE IF EXISTS shipments;

CREATE TABLE orders (
    order_id INTEGER PRIMARY KEY,
    product TEXT,
    quantity INTEGER
);

CREATE TABLE shipments (
    shipment_id INTEGER PRIMARY KEY,
    product TEXT,
    quantity INTEGER,
    order_id INTEGER
);

SELECT order_id, product, quantity FROM orders WHERE (product, quantity) IN (SELECT product, quantity FROM orders INTERSECT SELECT product, quantity FROM shipments);

SELECT order_id, product, quantity FROM orders WHERE (product, quantity) IN (SELECT product, quantity FROM orders INTERSECT SELECT product, quantity FROM shipments) ORDER BY product ASC;

SELECT product FROM orders EXCEPT SELECT product FROM shipments;

SELECT product FROM shipments EXCEPT SELECT product FROM orders ORDER BY product ASC;

SELECT order_id FROM orders WHERE order_id >= 10 EXCEPT SELECT order_id FROM shipments WHERE order_id >= 10;

SELECT order_id FROM shipments WHERE quantity < 5 EXCEPT SELECT order_id FROM orders WHERE quantity < 5;
